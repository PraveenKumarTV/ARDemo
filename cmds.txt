#!/bin/bash

# Update package list
sudo apt update

# Install KVM and virtualization tools
sudo apt install -y qemu-kvm libvirt-daemon-system libvirt-clients virt-manager

# Enable and start libvirt
sudo systemctl enable --now libvirtd

# Add current user to KVM and libvirt groups
sudo usermod -aG kvm "$USER"
sudo usermod -aG libvirt "$USER"

# Check KVM
echo ""
echo "Checking KVM..."
if [ -e /dev/kvm ]; then
    echo "KVM is available."
else
    echo "KVM is NOT available."
    echo "Enable Intel VT-x / AMD-V in BIOS/UEFI."
fi

echo ""
echo "KVM installation completed."
echo "Log out and log back in for group changes to take effect."
echo "Then run: virt-manager"

#!/bin/bash

VM_NAME="ubuntu-vm"
RAM="4096"
CPUS="2"
DISK_SIZE="25G"
ISO=$(ls ~/Downloads/*.iso | head -n 1)

echo "Creating VM: $VM_NAME"
echo "ISO: $ISO"

sudo virt-install \
  --name "$VM_NAME" \
  --memory "$RAM" \
  --vcpus "$CPUS" \
  --disk size=25 \
  --cdrom "$ISO" \
  --os-variant ubuntu24.04 \
  --network network=default \
  --graphics spice

echo "VM created."
Save and exit.
